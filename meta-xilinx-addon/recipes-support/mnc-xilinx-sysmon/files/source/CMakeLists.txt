cmake_minimum_required(VERSION 3.21)
project(MncXilinxSysmon VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(CTest)
add_library(mnc-xilinx-sysmon SHARED common/mnc/xilinx/sysmon/sysmon.cpp)
add_library(mnc::xilinx-sysmon ALIAS mnc-xilinx-sysmon)
set_target_properties(mnc-xilinx-sysmon PROPERTIES EXPORT_NAME xilinx-sysmon VERSION ${PROJECT_VERSION} SOVERSION 1)
target_compile_features(mnc-xilinx-sysmon PUBLIC cxx_std_23)
target_compile_options(mnc-xilinx-sysmon PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(mnc-xilinx-sysmon PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/common> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
install(TARGETS mnc-xilinx-sysmon EXPORT MncXilinxSysmonTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY common/mnc DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.hpp")
install(EXPORT MncXilinxSysmonTargets NAMESPACE mnc:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MncXilinxSysmon)
configure_package_config_file(MncXilinxSysmonConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MncXilinxSysmonConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MncXilinxSysmon)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MncXilinxSysmonConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MncXilinxSysmonConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MncXilinxSysmonConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MncXilinxSysmon)
if(BUILD_TESTING)
  add_executable(xilinx_sysmon_test tests/sysmon_test.cpp)
  target_link_libraries(xilinx_sysmon_test PRIVATE mnc::xilinx-sysmon)
  target_compile_options(xilinx_sysmon_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME xilinx_sysmon_test COMMAND xilinx_sysmon_test)
endif()
