cmake_minimum_required(VERSION 3.21)
project(MncSystem VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
option(MNC_SYSTEM_BUILD_DAEMON "Build privileged system manager" OFF)
option(MNC_SYSTEM_REQUIRE_SYSTEMD "Require native systemd support" OFF)
find_package(Threads REQUIRED)
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(SYSTEMD QUIET IMPORTED_TARGET libsystemd)
endif()
if(MNC_SYSTEM_REQUIRE_SYSTEMD AND NOT TARGET PkgConfig::SYSTEMD)
    message(FATAL_ERROR "libsystemd is required")
endif()
if(NOT TARGET glaze::glaze)
    if(MNC_GLAZE_SOURCE_DIR)
        add_subdirectory("${MNC_GLAZE_SOURCE_DIR}" glaze)
    else()
        find_package(glaze CONFIG REQUIRED)
    endif()
endif()
add_library(mnc_logging STATIC common/mnc/logging/logging.cpp common/mnc/logging/journal_reader.cpp)
add_library(mnc::logging ALIAS mnc_logging)
set_target_properties(mnc_logging PROPERTIES EXPORT_NAME logging)
add_library(mnc_service STATIC common/mnc/service/service.cpp common/mnc/service/service_manager.cpp)
add_library(mnc::service ALIAS mnc_service)
set_target_properties(mnc_service PROPERTIES EXPORT_NAME service)
target_link_libraries(mnc_service PUBLIC mnc_logging Threads::Threads)
add_library(mnc_system STATIC common/mnc/system/validation.cpp)
add_library(mnc::system ALIAS mnc_system)
set_target_properties(mnc_system PROPERTIES EXPORT_NAME system)
add_library(mnc_os_system STATIC common/mnc/os/system/client.cpp)
add_library(mnc::os-system ALIAS mnc_os_system)
set_target_properties(mnc_os_system PROPERTIES EXPORT_NAME os-system)
target_link_libraries(mnc_os_system PUBLIC mnc_system PRIVATE $<BUILD_INTERFACE:glaze::glaze>)
foreach(t mnc_logging mnc_service mnc_system mnc_os_system)
    target_include_directories(${t} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/common> $<INSTALL_INTERFACE:include>)
    target_compile_features(${t} PUBLIC cxx_std_23)
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
set(MNC_SYSTEM_WITH_SYSTEMD OFF)
if(TARGET PkgConfig::SYSTEMD)
    set(MNC_SYSTEM_WITH_SYSTEMD ON)
    foreach(t mnc_logging mnc_service mnc_os_system)
        target_link_libraries(${t} PRIVATE PkgConfig::SYSTEMD)
    endforeach()
    target_compile_definitions(mnc_logging PRIVATE MNC_LOGGING_HAVE_SYSTEMD=1)
    target_compile_definitions(mnc_service PRIVATE MNC_SERVICE_HAVE_SYSTEMD=1)
    target_compile_definitions(mnc_os_system PRIVATE MNC_SYSTEM_HAVE_SYSTEMD=1)
endif()
if(MNC_SYSTEM_BUILD_DAEMON)
    if(NOT TARGET PkgConfig::SYSTEMD)
        message(FATAL_ERROR "daemon needs libsystemd")
    endif()
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    add_library(mnc_system_backend STATIC daemon/backend.cpp daemon/storage.cpp daemon/timezone_catalog.cpp daemon/network_config.cpp daemon/ptp.cpp)
    target_include_directories(mnc_system_backend PUBLIC daemon)
    target_link_libraries(mnc_system_backend PUBLIC mnc_system glaze::glaze PkgConfig::SYSTEMD OpenSSL::Crypto)
    target_compile_options(mnc_system_backend PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(mnc-system-manager daemon/main.cpp)
    target_compile_options(mnc-system-manager PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(mnc-system-manager PRIVATE mnc_system_backend mnc_service)
    if(BUILD_TESTING)
        add_executable(mnc_ptp_test tests/ptp_test.cpp)
        target_link_libraries(mnc_ptp_test PRIVATE mnc_system_backend)
        target_compile_options(mnc_ptp_test PRIVATE -UNDEBUG)
        add_test(NAME mnc_ptp_test COMMAND mnc_ptp_test)
        add_executable(mnc_catalog_network_test tests/catalog_network_test.cpp)
        target_link_libraries(mnc_catalog_network_test PRIVATE mnc_system_backend)
        add_test(NAME mnc_catalog_network_test COMMAND mnc_catalog_network_test)
        add_executable(mnc_system_backend_test tests/backend_test.cpp)
        target_link_libraries(mnc_system_backend_test PRIVATE mnc_system_backend)
        add_test(NAME mnc_system_backend_test COMMAND mnc_system_backend_test)
    endif()
    install(TARGETS mnc-system-manager RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS mnc_logging mnc_service mnc_system mnc_os_system EXPORT MncSystemTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY common/mnc DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.hpp" PATTERN "protocol.hpp" EXCLUDE)
install(EXPORT MncSystemTargets NAMESPACE mnc:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MncSystem)
include(CMakePackageConfigHelpers)
configure_package_config_file(MncSystemConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MncSystemConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MncSystem)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MncSystemConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MncSystem)
if(BUILD_TESTING)
    add_executable(mnc_system_test tests/system_test.cpp)
    target_link_libraries(mnc_system_test PRIVATE mnc_system)
    target_compile_options(mnc_system_test PRIVATE -UNDEBUG)
    add_test(NAME mnc_system_test COMMAND mnc_system_test)
endif()
